local gameStartTime = 0

local gameEnterAt   = 0

local READY_DELAY   = 5.0

local function espReady()
    if consumePurge then consumePurge() end
    if not espReady() then return false end
    if gameEnterAt == 0 then return false end
    return (os.clock() - gameEnterAt) >= READY_DELAY
end
local lastCheck     = 0

local function safeCall(fn)
    local ok, v = pcall(fn)
    if ok then return v end
    return nil
end

local function espReady()
    if not _G.PlayerESP then return false end
    local now = os.clock()
    if now - lastCheck >= 1.0 then
        lastCheck = now
        local ship = safeCall(function() return SS_Field:GetStatic() end)
        local inGame = ship ~= nil and ship ~= 0
        if not gameStarted and inGame then
            gameStarted   = true
            gameEnterAt   = os.clock()
            gameStartTime = now
        elseif gameStarted and not inGame then
            gameStarted = false
            gameEnterAt = 0

            needPurge = true
            for k in pairs(deadPC) do deadPC[k] = nil end
        end
    end
    if not gameStarted then return false end
    return (os.clock() - gameStartTime) >= 5.0
end

local platformResult = {}
local platformNext   = {}
local platformTries  = {}

local PLATFORM_RETRY_DELAY = 1.0
local PLATFORM_MAX_TRIES   = 8

local function resolvePlatform(info)
    return safeCall(function()
        local client = AUCInst:GetStatic()
        if not client or client == 0 then return nil end
        local cd = GetClientFromPI:Call(client, info)
        if not cd or cd == 0 then return nil end
        local pd = BOMBA.read_ptr(cd, CD_PLATFORMDATA_OFF)
        if not pd or pd == 0 then return nil end
        local pid = BOMBA.read_int(pd, PSD_PLATFORM_OFF)
        if pid == nil then return nil end
        return PlatformNames[pid] or ("P" .. tostring(pid))
    end)
end

local function tryPlatform(key, info)
    local res = platformResult[key]
    if type(res) == "string" then return res end
    if (platformTries[key] or 0) >= PLATFORM_MAX_TRIES then return nil end
    local now = os.clock()
    if (platformNext[key] or 0) > now then return nil end
    platformTries[key] = (platformTries[key] or 0) + 1
    platformNext[key]  = now + PLATFORM_RETRY_DELAY
    local p = resolvePlatform(info)
    if p then platformResult[key] = p end
    return p
end

local function purgeKey(key)
    cache[key]          = nil
    platformResult[key] = nil
    platformNext[key]   = nil
    platformTries[key]  = nil
    npiJoinTime[key]    = nil
end

local function purgeAll()
    for k in pairs(cache) do cache[k] = nil end
    for k in pairs(pcPtrToKey) do pcPtrToKey[k] = nil end
    invalidateLocalCache()
end

local function fullReset()
    for k in pairs(cache) do cache[k] = nil end
    for k in pairs(labelSnap) do labelSnap[k] = nil end
    snapCount = 0
    for k in pairs(platformResult) do platformResult[k] = nil end
    for k in pairs(platformNext) do platformNext[k] = nil end
    for k in pairs(platformTries) do platformTries[k] = nil end
    for k in pairs(pcPtrToKey) do pcPtrToKey[k] = nil end
    for k in pairs(npiJoinTime) do npiJoinTime[k] = nil end
    for k in pairs(deadPC) do deadPC[k] = nil end
    for k in pairs(npiDead) do npiDead[k] = nil end
    invalidateLocalCache()
end

consumePurge = function()
    if needPurge then
        needPurge = false
        fullReset()
    end
end

local function killCd(info)
    if not info or info == 0 then return nil end
    if npiDead[tostring(info)] then return nil end
    return safeCall(function()
        local role = BOMBA.read_ptr(info, NPI_ROLE_OFF)
        if not role or role == 0 then return nil end
        local canKill = BOMBA.read_bool(role, RB_CANKILL_OFF)
        if not canKill then return nil end

        local pc = nil
        if NPI_GetObject then
            pc = NPI_GetObject:Call(info)
        end
        if not pc or pc == 0 then
            pc = BOMBA.read_ptr(info, NPI_OBJECT_OFF)
        end
        if not pc or pc == 0 then return nil end
        local t = BOMBA.read_float(pc, PC_KILLTIMER_OFF)
        if not t then return nil end
        if t < 0 then t = 0 end
        return t
    end)
end

local function colorOf(info)
    if not NPI_GetOutfit or not info or info == 0 then return nil end
    if npiDead[tostring(info)] then return nil end
    local cid = safeCall(function()
        local o = NPI_GetOutfit:Call(info)
        if not o or o == 0 then return nil end
        return BOMBA.read_int(o, OUTFIT_COLOR_OFF)
    end)

local function buildLabel(info, name, level, platform)
    if info and info ~= 0 and npiDead[tostring(info)] then
        return name or "?"
    end
    local roleName, isImpostor = nil, false
    local rolePtr = safeCall(function() return BOMBA.read_ptr(info, NPI_ROLE_OFF) end)
    if rolePtr and rolePtr ~= 0 then
        local rn = safeCall(function() return getNiceName:Call(rolePtr) end)
        if rn and rn ~= "" then
            roleName   = rn
            isImpostor = safeCall(function() return getIsImpostor:Call(rolePtr) end) or false
        end
    end
    local O = _G.PlayerESPOpt

    local wantLv  = O.showLevel and true or false
    local wantDev = O.showDevice and platform and true or false

    local line1 = ""
    if wantLv and wantDev then
        line1 = string.format("<color=#FFA500>Lv:%d - %s</color>", level, platform)
    elseif wantLv then
        line1 = string.format("<color=#FFA500>Lv:%d</color>", level)
    elseif wantDev then
        line1 = string.format("<color=#FFA500>%s</color>", platform)
    end

    local line2 = ""
    if roleName and O.showRole then
        line2 = isImpostor
            and string.format("<color=#FF4444>%s</color>", roleName)
            or  string.format("<color=#00FFFF>%s</color>", roleName)
    end

    local cdLine = ""
    if O.showKillCd and isLocalPlayer(info) then
        local cd = killCd(info)
        if cd then
            if cd > 0.05 then
                cdLine = string.format("<color=#FF4444>CD %.0fs</color>", cd)
            else
                cdLine = "<color=#00FF66>KILL READY</color>"
            end
        end
    end

    local shown = name
    if O.colorName then
        local hex = colorOf(info)
        if hex then
            shown = "<color=" .. hex .. ">" .. name .. "</color>"
        end
    end

    local out = {}
    if line1 ~= "" then out[#out+1] = line1 end
    if line2 ~= "" then out[#out+1] = line2 end
    if cdLine ~= "" then out[#out+1] = cdLine end
    out[#out+1] = shown
    return table.concat(out, "\n")
end

local function getCachedLabel(info)
    local key = tostring(info)

    local joinTime = npiJoinTime[key]
    if joinTime and (os.clock() - joinTime) < 3.0 then return nil end

    local entry = cache[key]
    if entry then
        entry.ticks = entry.ticks + 1
        if entry.platform == nil then
            local p = tryPlatform(key, info)
            if p then
                entry.platform = p
                entry.label    = buildLabel(info, entry.name, entry.level, p)
                labelSnap[key] = entry.label
            end
        end

    if _G.PlayerESPOpt.showKillCd then
        entry.label = buildLabel(info, entry.name, entry.level, entry.platform)
        labelSnap[key] = entry.label
        return entry.label
    end
    if entry.ticks < CACHE_TTL then return entry.label end
        entry.ticks = 0
        local newRolePtr = safeCall(function() return BOMBA.read_ptr(info, NPI_ROLE_OFF) end)
        if newRolePtr ~= entry.rolePtr then
            entry.rolePtr = newRolePtr
            entry.label   = buildLabel(info, entry.name, entry.level, entry.platform)
            labelSnap[key] = entry.label
        end
        return entry.label
    end

    local name = safeCall(function() return getPlayerName:Call(info) end)
    if not name or name == "" then return nil end

    local level = safeCall(function()
        local v = playerLevelField:Get(info)
        return v and (v + 1) or 1
    end) or 1

    local rolePtr  = safeCall(function() return BOMBA.read_ptr(info, NPI_ROLE_OFF) end)
    local platform = tryPlatform(key, info)
    local label    = buildLabel(info, name, level, platform)

    cache[key] = {
        label = label, name = name, level = level,
        platform = platform, rolePtr = rolePtr, ticks = 0
    }

    local pcPtr = safeCall(function() return BOMBA.read_ptr(info, NPI_OBJECT_OFF) end)
    if pcPtr and pcPtr ~= 0 then pcPtrToKey[tostring(pcPtr)] = key end
    if labelSnap[key] == nil then snapCount = snapCount + 1 end
    labelSnap[key] = label
    return label
end

local function applyLabel(info, textObject)
    if not info or not textObject then return end
    if isLocalPlayer(info) and not _G.PlayerESPOpt.showSelf then return end
    local text = getCachedLabel(info)
    if text then safeCall(function() setText:Call(textObject, text) end) end
end

local pcOnDestroy = PlayerControl:GetMethod("OnDestroy", 0)
BOMBA.HOOK(pcOnDestroy, {ret="void",args={"ptr"}}, function(instance)
    safeCall(function()
        if not instance or instance == 0 then return end
        local pcKey   = tostring(instance)

        deadPC[pcKey]  = true
        npiDead[pcKey] = true

        local infoKey = pcPtrToKey[pcKey]
        if infoKey then
            purgeKey(infoKey)
            npiDead[infoKey] = true
            pcPtrToKey[pcKey] = nil
        end

        for i = #pendingVotes, 1, -1 do
            local v = pendingVotes[i]
            if v and (v.key == pcKey or v.key == infoKey) then
                table.remove(pendingVotes, i)
            end
        end

        if #pendingVotes > 0 then
            for i = #pendingVotes, 1, -1 do pendingVotes[i] = nil end
        end

        invalidateLocalCache()
    end)
    return BOMBA.call_original(instance)
end)

local meetingUpdate = MeetingHud:GetMethod("Update", 0)
BOMBA.HOOK(meetingUpdate, {ret="void",args={"ptr"}}, function(instance)
    local result = BOMBA.call_original(instance)
    if not _G.PlayerESP then return result end
    if not _G.PlayerESPOpt.showMeeting then return result end
    if #pendingVotes == 0 then return result end

    if deadPC[tostring(instance)] then return result end

    safeCall(function()

        local arr = MH_PlayerStates and safeCall(function()
            return MH_PlayerStates:Get(instance) end) or nil
        if not arr or arr == 0 then return end
        local alen = safeCall(function()
            return BOMBA.read_int(arr, 0x18) end)
        if not alen or alen <= 0 or alen > 32 then return end

        for i = 1, #pendingVotes do
            local v = pendingVotes[i]
            if not v.done then
                local area = nil
                if v.idx and v.idx < alen then
                    area = safeCall(function()
                        return BOMBA.read_ptr(arr, 0x20 + v.idx * 8) end)
                end
                if v.isLocal or not area or area == 0 then
                    v.done = true
                else

                    local label = v.label or labelSnap[v.key]
                    if label then
                        local textObject = safeCall(function()
                            return voteNameTextField:Get(area)
                        end)
                        if textObject and textObject ~= 0 then
                            local ok = safeCall(function()
                                setText:Call(textObject, label)
                                return true
                            end)
                            if ok then v.done = true end
                        end
                    else
                        v.done = true
                    end
                end
            end
        end
    end)
    return result
end)

local function tryMethod(cls, n, argc, sig)
    if not cls then return nil end
    local ok, m = pcall(function()
        local mm = cls:GetMethod(n, argc)
        if not mm then return nil end
        if sig then return mm:cast(sig) end
        return mm
    end)
    if ok then return m end
    return nil
end

    local ok, f = pcall(function()
        local ff = cls:GetField(n)
        if not ff then return nil end
        if t then return ff:cast(t) end
        return ff
    end)
    if ok then return f end
    return nil
end

local function charAt(i)
    return string.sub(V2CHARS, i + 1, i + 1)
end

local function codeFromInt(num)
    if not num then return "" end
    num = math.floor(num)

    if num < 0 then num = num + 4294967296 end
    local a = num % 1024
    local b = math.floor(num / 1024) % 1048576
    return charAt(a % 26)
        .. charAt(math.floor(a / 26) % 26)
        .. charAt(b % 26)
        .. charAt(math.floor(b / 26) % 26)
        .. charAt(math.floor(b / 676) % 26)
        .. charAt(math.floor(b / 17576) % 26)
end

local function refreshInfo()

    if S.show.alive or S.show.players then
        iAlive, iTotal = 0, 0
        local gd = GD_Field and safeCall(function() return GD_Field:GetStatic() end) or nil
        if gd and gd ~= 0 and GD_GetCount then
            local n = safeCall(function() return GD_GetCount:Call(gd) end) or 0
            iTotal = n
        end
    end

    if S.show.code and AUC_Field and AUC_GameId then
        local auc = safeCall(function() return AUC_Field:GetStatic() end)
        if auc and auc ~= 0 then
            local id = safeCall(function() return AUC_GameId:Get(auc) end)
            if id and id ~= 0 then

                iCode = (ok and c) or ""
            else

            iHost = h and "HOST" or "GUEST"

    local nowIn = ship ~= nil and ship ~= 0
    if nowIn and not inGame then gameStart = os.clock() end

local function buildInfoLine()
    local p = {}
    if S.show.code   and iCode  ~= "" then p[#p+1] = iCode end
    if S.show.host   and iHost  ~= "" then p[#p+1] = iHost end
    if S.show.players and iTotal > 0  then p[#p+1] = iTotal .. "p" end
    if S.show.role   and iRole  ~= "" then p[#p+1] = iRole end
    if S.show.level  and iLevel ~= "" then p[#p+1] = iLevel end
    if S.show.fps                     then p[#p+1] = fpsVal .. "fps" end
    if S.show.clock                   then p[#p+1] = os.date("%H:%M") end
    if S.show.timer and inGame then
        local t = math.floor(os.clock() - gameStart)
        p[#p+1] = string.format("%d:%02d", math.floor(t / 60), t % 60)
    end
    if #p == 0 then return "" end
    return table.concat(p, " | ")
end

local function stepper(label, value, step, lo, hi)
    ImGui.Text(label .. ": " .. tostring(value))
    if ImGui.Button("-##" .. label, { 46, 26 }) then
        value = value - step
        if value < lo then value = lo end
    end
    ImGui.SameLine()
    if ImGui.Button("+##" .. label, { 46, 26 }) then
        value = value + step
        if value > hi then value = hi end
    end
    ImGui.SameLine()
    if ImGui.Button("0##" .. label, { 46, 26 }) then
        value = 0
    end
    return value
end

                    platform = BOMBA.read_int(pd, 0x10) or 0
                end)
            end

            local pid = safeCall(function()
                return BOMBA.read_int(info, 0x35) end)
            local isDead = safeCall(function()
                return BOMBA.read_bool(info, 0x78) end)

            out[#out+1] = {
                name = name, colorId = colorId,
                level = readLevel(info), roleName = readRoleName(info),
                platform = platform,
                playerId = pid and (pid % 256) or -1,
                dead = isDead and true or false
            }
        end
    end
    return out
end

local function splitMsgs(head, items)
    local msgs, cur = {}, nil
    for i = 1, #items do
        local p = items[i]
        if cur == nil then
            cur = head .. p
        elseif #cur + 3 + #p <= MAX_CHAT_LEN then
            cur = cur .. " | " .. p
        else
            msgs[#msgs+1] = cur
            cur = head .. p
        end
    end
    if cur then msgs[#msgs+1] = cur end
    return msgs
end

local function buildStats(colorName)
    local cid = COLOR_MAP[string.lower(tostring(colorName or ""))]
    if cid == nil then return { "Unknown color: " .. tostring(colorName) } end

    local players = getAllPlayers()
    for i = 1, #players do
        local p = players[i]
        if p.colorId == cid then
            return { p.name .. " | Lv:" .. formatLevel(p.level)
                .. " | " .. p.roleName .. " | " .. (PLATFORMS[p.platform] or "?") }
        end
    end
    return { "Player not found: " .. tostring(colorName) }
end

local IMP_ROLES = { "impostor", "shapeshifter", "phantom", "viper" }

local function isImpRole(roleName)
    local r = string.lower(tostring(roleName or ""))
    for i = 1, #IMP_ROLES do
        if string.find(r, IMP_ROLES[i], 1, true) then return true end
    end
    return false
end

local function buildRoles(query)
    local q = string.lower(tostring(query or ""))
    if ALIAS[q] then q = ALIAS[q] end
    if q == "" then return { "Usage: ?roles {role}" } end

    local players = getAllPlayers()
    if #players == 0 then return { "No player data" } end

    local allImp = (q == "all imp" or q == "allimp" or q == "all impostor"
                 or q == "imps" or q == "all imps")

    local found = {}
    for i = 1, #players do
        local p = players[i]
        local match
        if allImp then
            match = isImpRole(p.roleName)
        else
            match = string.find(string.lower(p.roleName), q, 1, true)
        end
        if match then
            local label = p.name .. " (" .. (COLOR_NAMES[p.colorId] or "?") .. ")"
            if allImp then
                label = label .. " " .. p.roleName
            end
            found[#found+1] = label
        end
    end

    if #found == 0 then
        if allImp then return { "No impostor found" } end
        return { "No one with role: " .. tostring(query) }
    end

    if allImp then return splitMsgs("Imps: ", found) end
    return splitMsgs(query .. ": ", found)
end

local function InitializeHat()

    if hatInitialized then
        return true
    end

    local ok, err =
        pcall(function()

            HatPlayerControl =
                BOMBA.Class(
                    "",
                    "PlayerControl"
                )

            if not HatPlayerControl then
                error("PlayerControl no encontrado")
            end

            HatRpcSetHat =
                HatPlayerControl:GetMethod(
                    "RpcSetHat",
                    1
                )

            if not HatRpcSetHat then
                error("RpcSetHat no encontrado")
            end

            HatRpcSetHat =
                HatRpcSetHat:cast({
                    ret = "void",
                    args = {
                        "ptr",
                        "string"
                    }
                })

            HatLocalPlayerField =
                HatPlayerControl:GetField(
                    "LocalPlayer"
                )

            if not HatLocalPlayerField then
                error("LocalPlayer no encontrado")
            end

            HatLocalPlayerField =
                HatLocalPlayerField:cast("ptr")

            HatFixedUpdate =
                HatPlayerControl:GetMethod(
                    "FixedUpdate",
                    0
                )

            if not HatFixedUpdate then
                error("FixedUpdate no encontrado")
            end

        end)

    if not ok then

        SayLog(
            "[ChangerHat] ERROR: " ..
            tostring(err)
        )

        hatInitialized = false

        return false

    end

    hatInitialized = true

    return true

end

local function GetHatLocalPlayer()

    if not HatLocalPlayerField then
        return nil
    end

    local ok, player =
        pcall(function()

            return HatLocalPlayerField:GetStatic()

        end)

    if not ok then
        return nil
    end

    return player

end

local function QueueNextHat()

    if not hatInitialized then
        return
    end

    if executingHat then
        return
    end

    if #Hats == 0 then
        return
    end

    currentHatIndex =
        currentHatIndex + 1

    if currentHatIndex > #Hats then
        currentHatIndex = 1
    end

    pendingHat = true

end

local function ProcessPendingHat()

    if not pendingHat then
        return
    end

    if executingHat then
        return
    end

    pendingHat = false
    executingHat = true

    local hat =
        Hats[currentHatIndex]

    if not hat then
        executingHat = false
        return
    end

    local player =
        GetHatLocalPlayer()

    if not player then

        executingHat = false

        SayLog(
            "[ChangerHat] LocalPlayer nil"
        )

        return

    end

    local ok, err =
        pcall(function()

            HatRpcSetHat:Call(
                player,
                hat.id
            )

        end)

    if not ok then

        SayLog(
            "[ChangerHat] RpcSetHat ERROR: " ..
            tostring(err)
        )

    else

        SayLog(
            "[ChangerHat] Hat enviado: " ..
            tostring(hat.id)
        )

    end

    executingHat = false

end

local function UpdateHatChanger()

    if not changerHat then
        return
    end

    if pendingHat then
        return
    end

    if executingHat then
        return
    end

    hatTimer =
        hatTimer + 0.02

    if hatTimer >= HAT_INTERVAL then

        hatTimer = 0

        QueueNextHat()

    end

end

local function SendPet(petId, petName)

    if not petId or petId == "" then

        petStatus = "ERROR"
        petErrorText = "Pet ID vacío"

        return

    end

    petPending = true
    petPendingId = petId

    petCurrentName =
        petName or "Desconocido"

    petCurrentId = petId

    petStatus = "PENDING"
    petErrorText = ""

end

local function ProcessPendingPet()

    if not petPending then
        return
    end

    petPending = false

    if not petPendingId or
       petPendingId == "" then

        petStatus = "ERROR"
        petErrorText = "Pet ID inválido"

        return

    end

    local petId =
        petPendingId

    petPendingId = nil

    local localPlayer = nil

    local getPlayerOK,
          getPlayerError =
        pcall(function()

            localPlayer =
                PetLocalPlayerField:GetStatic()

        end)

    if not getPlayerOK or
       not localPlayer then

        petStatus = "ERROR"

        petErrorText =
            "LocalPlayer no encontrado: " ..
            tostring(getPlayerError)

        return

    end

    local localOK,
          localError =
        pcall(function()

            PetRawSetPet:Call(
                localPlayer,
                petId,
                0
            )

        end)

    if not localOK then

        petStatus = "ERROR"

        petErrorText =
            "RawSetPet: " ..
            tostring(localError)

        return

    end

    local rpcOK,
          rpcError =
        pcall(function()

            PetRpcSetPet:Call(
                localPlayer,
                petId
            )

        end)

    if rpcOK then

        petStatus = "OK - RPC"
        petErrorText = ""

    else

        petStatus = "ERROR RPC"

        petErrorText =
            "RpcSetPet: " ..
            tostring(rpcError)

    end

end

local function ProcessAutoPet()

    if not changerPet or
       #Pets == 0 then

        return

    end

    petTimer =
        petTimer + 0.02

    if petTimer >= PET_INTERVAL then

        petTimer = 0

        petCurrentIndex =
            petCurrentIndex + 1

        if petCurrentIndex > #Pets then
            petCurrentIndex = 1
        end

        SendPet(
            PetIDs[petCurrentIndex],
            Pets[petCurrentIndex]
        )

    end

end

            lastError = ""

        lastError = ""

        visor.name

        visor.id

local function BindOutfit()

    if outfitBound then
        return OutfitLPField ~= nil
    end

    outfitBound = true

    pcall(function()

        OutfitPC = BOMBA.Class("", "PlayerControl")

        OutfitLPField =
            OutfitPC:GetField("LocalPlayer"):cast("ptr")

        OutfitGetData =
            OutfitPC:GetMethod("get_Data", 0)
                :cast({ ret = "ptr", args = { "ptr" } })

        local npi = BOMBA.Class("", "NetworkedPlayerInfo")

        OutfitGetOutfit =
            npi:GetMethod("get_DefaultOutfit", 0)
                :cast({ ret = "ptr", args = { "ptr" } })

        local outfitClass = nil

        local candidates = {
            { "NetworkedPlayerInfo", "PlayerOutfit" },
            { "", "PlayerOutfit" },
            { "", "NetworkedPlayerInfo/PlayerOutfit" },
        }

        for i = 1, #candidates do

            if not outfitClass then

                local okc, c = pcall(function()
                    return BOMBA.Class(
                        candidates[i][1],
                        candidates[i][2]
                    )
                end)

                if okc and c then
                    outfitClass = c
                end

            end

        end

        if outfitClass then

            local fields = {}
            local found  = 0

            local names = { "HatId", "SkinId", "PetId", "VisorId" }

            for i = 1, #names do

                local okf, f = pcall(function()
                    local ff = outfitClass:GetField(names[i])
                    if not ff then return nil end
                    return ff:cast("string")
                end)

                if okf and f then
                    fields[names[i]] = f
                    found = found + 1
                end

            end

            if found == 4 then
                OutfitFields = fields
            else
                SayLog(
                    "[save] PlayerOutfit incomplet: " ..
                    tostring(found) .. "/4"
                )
            end

        else

            SayLog("[save] classe PlayerOutfit introuvable")

        end

    end)

    return OutfitLPField ~= nil

end

local function GetLocalOutfit()

    if not OutfitLPField or not OutfitGetData
       or not OutfitGetOutfit or not OutfitFields then
        return nil
    end

    local outfit = nil

    pcall(function()

        local lp = OutfitLPField:GetStatic()

        if not lp or lp == 0 then
            return
        end

        if not OutfitGetData then
            return
        end

        local data = OutfitGetData:Call(lp)

        if not data or data == 0 then
            return
        end

        if not OutfitGetOutfit then
            return
        end

        local o = OutfitGetOutfit:Call(data)

        if o and o ~= 0 then
            outfit = o
        end

    end)

    return outfit
end

local function ReadCurrentOutfit()

    local outfit = GetLocalOutfit()

    if not outfit then
        return nil
    end

    if not OutfitFields then
        return nil
    end

    local res = {}

    local names = { "HatId", "SkinId", "PetId", "VisorId" }

    for i = 1, #names do

        local key = names[i]
        local val = ""

        local f = OutfitFields[key]

        if f then

            local ok, v = pcall(function()
                return f:Get(outfit)
            end)

            if ok and v and v ~= "" then
                val = tostring(v)
            end

        end

        res[key] = val

    end

    if res.HatId == "" and res.SkinId == ""
       and res.PetId == "" and res.VisorId == "" then
        return nil
    end

    return {
        hatId   = res.HatId,
        skinId  = res.SkinId,
        petId   = res.PetId,
        visorId = res.VisorId,
    }
end

local function scan()

    if not setup() then
        return
    end

    local data = GD_InstanceField:GetStatic()

    if data == nil or data == 0 then
        Guard.status = "Waiting GameData"
        return
    end

    local list = GD_AllPlayersField:Get(data)

    if list == nil or list == 0 then
        Guard.status = "Waiting players"
        return
    end

    if not ListGetItem then

        ListGetItem = BOMBA.Class(list):GetMethod("get_Item", 1)

        if not ListGetItem then
            Guard.status = "get_Item failed"
            return
        end

        ListGetItem = ListGetItem:cast({
            ret = "ptr",
            args = {"ptr", "int"}
        })
    end

    local count = tonumber(GD_GetPlayerCount:Call(data)) or 0

    local found = {}
    local hostNow = amHost()

    Guard.successfulReads = 0
    Guard.failedReads = 0

    for i = 0, count - 1 do

        local info = ListGetItem:Call(list, i)

        if info ~= nil and info ~= 0 then

            local player = {
                name = "Unknown",
                level = 0,
                clientId = -1
            }

            if PlayerLevelField then
                player.level = tonumber(PlayerLevelField:Get(info)) or 0
            end

            if ClientIdField then
                player.clientId = tonumber(ClientIdField:Get(info)) or -1
            end

            if PlayerNameMethod then
                local name = PlayerNameMethod:Call(info)
                if name then
                    player.name = tostring(name)
                end
            end

            found[#found + 1] = player
            Guard.successfulReads = Guard.successfulReads + 1

            if hostNow
                and Guard.enabled
                and player.clientId >= 0
                and not Guard.processed[player.clientId] then

                if player.level < GUI.minimumLevel
                    or player.level > GUI.maximumLevel then

                    log("Kicking " .. player.name ..
                        " (level " .. tostring(player.level) .. ")")

                    kickPlayer(player.clientId, GUI.banInsteadOfKick)
                    Guard.processed[player.clientId] = true
                else

                    Guard.processed[player.clientId] = true
                end
            end

        else
            Guard.failedReads = Guard.failedReads + 1
        end
    end

    Guard.players = found
    Guard.scanCount = Guard.scanCount + 1
    Guard.status = "OK " .. tostring(#found) .. " players"
end

local function update()

    if os.clock() >= Guard.nextScan then
        Guard.nextScan = os.clock() + Guard.interval
        scan()
    end
end

function AddLog(eventType, playerName, message, playerColor)
    if not Extra.logging or Extra.paused then return end
    table.insert(Extra.logs, {
        time        = os.date("%H:%M:%S"),
        type        = eventType,
        player      = playerName or "Unknown",
        msg         = message or "",
        playerColor = playerColor,
    })
    Extra.stats[eventType] = (Extra.stats[eventType] or 0) + 1
    while #Extra.logs > Extra.maxLogs do table.remove(Extra.logs, 1) end
end

local function posOf(pc)

local function snapTo(x, y)
    local lp = LP_Field and safeCall(function()
        return LP_Field:GetStatic() end) or nil
    if not lp or lp == 0 then return false end

    local done = false

    safeCall(function()
        local nt = BOMBA.read_ptr(lp, NT_OFF)
        if not nt or nt == 0 then return end
        BOMBA.write_float(nt, NT_POSX_OFF, x)
        BOMBA.write_float(nt, NT_POSY_OFF, y)
        local seq = BOMBA.read_int(nt, NT_SEQ_OFF)
        if seq then
            BOMBA.write_int(nt, NT_SEQ_OFF, (seq + 1) % 65536)
        end
    end)

    safeCall(function()
        local nt = BOMBA.read_ptr(lp, NT_OFF)
        if not nt or nt == 0 then return end
        local body = BOMBA.read_ptr(nt, NT_BODY_OFF)
        if not body or body == 0 then return end

        if not RB_MovePosition and Rigidbody2D then
            RB_MovePosition = tryMethod(Rigidbody2D, "set_position", 1,
                { ret = "void", args = { "ptr", "float", "float" } })
        end
        if RB_MovePosition then
            RB_MovePosition:Call(body, x, y)
            done = true
        end
    end)

    return done
end

local function refresh()
    for i = #players, 1, -1 do players[i] = nil end

    local gd = GD_Field and safeCall(function()
        return GD_Field:GetStatic() end) or nil
    if not gd or gd == 0 then
        status = "not in game" ; return
    end

    local list = GD_AllPlayers and safeCall(function()
        return GD_AllPlayers:Get(gd) end) or nil
    if not list or list == 0 then
        status = "no player list" ; return
    end

    resolveList(list)
    if not ListGetItem then
        status = "get_Item unresolved" ; return
    end

    local count = 0
    if GD_GetCount then
        count = tonumber(safeCall(function()
            return GD_GetCount:Call(gd) end)) or 0
    end
    if count <= 0 or count > 32 then
        status = "bad count" ; return
    end

    local me = LP_Field and safeCall(function()
        return LP_Field:GetStatic() end) or nil

    for i = 0, count - 1 do
        local info = safeCall(function() return ListGetItem:Call(list, i) end)
        if info and info ~= 0 then
            local pc = NPI_GetObj and safeCall(function()
                return NPI_GetObj:Call(info) end) or nil

            if pc and pc ~= 0 and pc ~= me then
                local nm = "?"
                if NPI_GetName then
                    nm = tostring(safeCall(function()
                        return NPI_GetName:Call(info) end) or "?")
                end
                local pid = safeCall(function()
                    return BOMBA.read_int(info, NPI_PID_OFF) end)

                players[#players + 1] = {
                    name = nm,
                    id = pid and (pid % 256) or -1,
                }
            end
        end
    end

    if S.target >= #players then S.target = 0 end
    status = "found " .. tostring(#players) .. " players"
end

local function tryClass(ns, n)
    local ok, c = pcall(function() return BOMBA.Class(ns, n) end)
    if ok and c then return c end
    return nil
end

local function tryMethod(cls, n, argc, sig)
    if not cls then return nil end
    local ok, m = pcall(function()
        local mm = cls:GetMethod(n, argc)
        if not mm then return nil end
        if sig then return mm:cast(sig) end
        return mm
    end)
    if ok then return m end
    return nil
end

local function tryField(cls, n, t)
    if not cls then return nil end
    local ok, f = pcall(function()
        local ff = cls:GetField(n)
        if not ff then return nil end
        if t then return ff:cast(t) end
        return ff
    end)
    if ok then return f end
    return nil
end

local function log(t)
    table.insert(logs, 1, tostring(t))

local function ship()
    if not SS_Field then return nil end
    local ss = safeCall(function() return SS_Field:GetStatic() end)
    if not ss or ss == 0 then return nil end
    return ss
end

local function amHost()
    if not AUC_Field or not AmHost then return false end
    local auc = safeCall(function() return AUC_Field:GetStatic() end)
    if not auc or auc == 0 then return false end
    return safeCall(function() return AmHost:Call(auc) end) == true
end

local function detectMap(ss)
    if not ss then return -1 end

    local sid = safeCall(function()
        return BOMBA.read_int(ss, SS_SPAWNID_OFF) end)
    if sid and MAPS[sid] then return sid end

    local t = safeCall(function()
        return BOMBA.read_int(ss, SS_TYPE_OFF) end)
    if t == 0 then return 0 end
    if t == 1 then return 1 end
    if t == 2 then return 2 end
    if t == 3 then return 5 end
    return -1
end

local function fire(ss, id, amount)
    if not ss or not RpcUpdateSystem then return false end
    local ok = safeCall(function()
        RpcUpdateSystem:Call(ss, id, amount)
        return true
    end)
    if ok then sent = sent + 1 end
    return ok == true
end

local function shutDoor(ss, id)
    if not ss or not RpcCloseDoors then return false end
    local ok = safeCall(function()
        RpcCloseDoors:Call(ss, id)
        return true
    end)
    if ok then sent = sent + 1 end
    return ok == true
end

local SwitchSystemCls = tryClass("", "SwitchSystem")
local lightsInst = nil

if SwitchSystemCls then
    local lightsUpdate = tryMethod(SwitchSystemCls, "UpdateSystem", 2)
    if lightsUpdate then
        pcall(function()
            BOMBA.HOOK(lightsUpdate,
                { ret = "void", args = { "ptr", "ptr", "ptr" } },
                function(instance, player, reader)
                    if instance and instance ~= 0 then
                        lightsInst = instance
                    end
                    return BOMBA.call_original(instance, player, reader)
                end)
        end)
    end
end

local function fixLights(ss)
    if not ss then return false end

    local current = nil
    if lightsInst then
        current = safeCall(function()
            return BOMBA.read_int(lightsInst, 0x14)
        end)
        if current then
            current = current % 256
            if current % 32 == 31 then
                return true
            end
        end
    end

    if fire(ss, 7, 16) then
        return true
    end

    if not lightsInst then
        return false
    end

    current = safeCall(function()
        return BOMBA.read_int(lightsInst, 0x14)
    end)
    if current then
        current = current % 256
        return current % 32 == 31
    end

    return false
end

if _G.HookBus then
    _G.HookBus.on("FixedUpdate", function(instance)
                local r = nil

                pcall(function()
                    host = amHost()
                    local ss = ship()

                    if not ss then
                        status  = "not in game"
                        mapId   = -1
                        mapName = nil
                        redStep = 0
                        return
                    end

                    mapId = detectMap(ss)
                    mapName = MAPS[mapId] and MAPS[mapId].name or nil
                    if not mapName then
                        status = "map not detected"
                        return
                    end

                    local M = MAPS[mapId]

                    while qSabs[1] do
                        local q = table.remove(qSabs, 1)
                        local okq
                        if q[2] == "fix" then
                            okq = fixLights(ss)
                        else
                            okq = fire(ss, q[1], q[2])
                        end
                        if okq then
                            status = "sab " .. tostring(q[1]) .. " sent"
                        else
                            status = "failed"
                        end
                        log(status)
                    end

                    if qDoor then
                        local id, label = qDoor[1], qDoor[2]
                        qDoor = nil
                        if shutDoor(ss, id) then
                            status = "door " .. label
                        else
                            status = "door failed"
                        end
                        log(status)
                    end

                    if wantAllDoors then
                        wantAllDoors = false
                        local n = 0
                        for i = 1, #M.doors do
                            if shutDoor(ss, M.doors[i][2]) then n = n + 1 end
                        end
                        status = tostring(n) .. " doors closed"
                        log(status)
                    end

                    if wantAll then
                        wantAll = false
                        local n = 0
                        for i = 1, #M.sab do
                            if fire(ss, M.sab[i][1], M.sab[i][3]) then
                                n = n + 1
                            end
                        end
                        status = "sabotage all: " .. tostring(n)
                        log(status)
                    end

                    if wantEnd then
                        wantEnd = false
                        endLoop = not endLoop
                        endTimer = 0
                        if endLoop then
                            status = "END loop ON"
                        else
                            status = "END loop OFF"
                        end
                        log(status)
                    end

                    if endLoop then
                        local now = os.clock()
                        if now - endTimer >= 5.0 then
                            endTimer = now
                            local d = 0
                            for i = 1, #M.doors do
                                if shutDoor(ss, M.doors[i][2]) then
                                    d = d + 1
                                end
                            end
                            fire(ss, SYS_REACTOR, 128)
                            fire(ss, SYS_LIFESUPP, 128)
                            status = "END: " .. tostring(d)
                                .. " doors + 2 sabs"
                            log(status)
                        end
                    end

                    if wantRed then
                        wantRed = false
                        fire(ss, SYS_LIFESUPP, 128)
                        redStep = 1
                        redAt   = os.clock() + S.redDelay
                        status  = "red: O2 sent"
                        log(status)
                    end

                    if redStep == 1 and os.clock() >= redAt then
                        redStep = 0
                        fire(ss, SYS_REACTOR, 128)
                        status = "red: reactor sent"
                        log(status)
                    end

                    if S.alwaysFix then
                        local now = os.clock()
                        if now - fixTimer >= 1.0 then
                            fixTimer = now
                            for i = 1, #M.sab do
                                local d = M.sab[i]
                                if S.fixSel[mapId .. "_" .. d[1]] ~= false then
                                    if d[1] == 7 then
                                        fixLights(ss)
                                    else
                                        fire(ss, d[1], 16)
                                    end
                                end
                            end
                            if S.fixDoors then
                                for i = 1, #M.doors do
                                    fire(ss, M.doors[i][2], 0)
                                end
                            end
                        end
                    end

                    if S.spamOn then
                        local now = os.clock()
                        if now - spamTimer >= S.spamDelay then
                            spamTimer = now
                            spamState = not spamState
                            local sd = M.sab[S.spamPick] or M.sab[1]
                            if sd then
                                fire(ss, sd[1], spamState and sd[3] or 16)
                            end
                        end
                    end
                end)

    end)
end

local function netTransform()
    if not LP_Field or not NT_Field then return nil end
    return safeCall(function()
        local lp = LP_Field:GetStatic()
        if not lp or lp == 0 then return nil end
        local nt = NT_Field:Get(lp)
        if not nt or nt == 0 then return nil end
        return nt
    end)
end

local function isMine(inst)
    if not inst or inst == 0 then return false end
    local nt = netTransform()
    return nt ~= nil and nt == inst
end

local function freeze(inst)
    if not inst or inst == 0 then return end
    safeCall(function()
        BOMBA.write_float(inst, POS_X,  S.savedX)
        BOMBA.write_float(inst, POS_Y,  S.savedY)
        BOMBA.write_float(inst, SENT_X, S.savedX)
        BOMBA.write_float(inst, SENT_Y, S.savedY)
        writes = writes + 1
    end)
end

            function(inst, writer, initial)
                pcall(function()
                    if S.enabled and isMine(inst) then freeze(inst) end
                end)
                return BOMBA.call_original(inst, writer, initial)
            end)
    end)
end

                    end

                            status = "off"

                            local y = BOMBA.read_float(nt, SENT_Y)

                                status = "cannot read position"
                            end
                        end

                        curX, curY = S.savedX, S.savedY
                    else
                        curX = BOMBA.read_float(nt, SENT_X) or 0.0
                        curY = BOMBA.read_float(nt, SENT_Y) or 0.0
                    end
                end)

                return r
            end)

local function infoOf(pc)

local function nameOf(info)
    if not info or not NPI_GetName then return "?" end
    local n = safeCall(function() return NPI_GetName:Call(info) end)
    if n then
        n = tostring(n)
        if n ~= "" then return n end
    end
    return "?"
end

local function colorOf(info)
    if not info or not NPI_GetOutfit then return -1 end
    local c = safeCall(function()
        local o = NPI_GetOutfit:Call(info)
        if not o or o == 0 then return nil end
        return BOMBA.read_int(o, OUTFIT_COLOR)
    end)
    return c or -1
end

local function pidOf(info)
    if not info then return -1 end
    local p = safeCall(function()
        return BOMBA.read_int(info, NPI_PLAYERID) end)
    if not p then return -1 end
    return p % 256
end

local function sourceContainer()
    local lp = LP_Field and safeCall(function()
        return LP_Field:GetStatic() end) or nil
    if not lp or lp == 0 then return nil end

    return safeCall(function()
        local cos = BOMBA.read_ptr(lp, COSMETICS_OFF)
        if not cos or cos == 0 then return nil end
        local cont = BOMBA.read_ptr(cos, NAMECONT_OFF)
        if not cont or cont == 0 then return nil end
        return cont
    end)
end

local function clockStr(sec)
    if sec < 0 then sec = 0 end
    return string.format("%02d:%02d",
        math.floor(sec / 60), math.floor(sec % 60))
end

local function labelFor(k, now)
    local vc = COLOR_HEX[k.victimColor]
    local kc = COLOR_HEX[k.killerColor]

    local v = vc and ("<color=" .. vc .. ">" .. k.victim .. "</color>")
        or k.victim
    local m = kc and ("<color=" .. kc .. ">" .. k.killer .. "</color>")
        or k.killer

    local txt = v .. "\nkilled by " .. m
    if S.showTime then
        txt = txt .. "\n" .. clockStr(now - k.at)
    end
    return txt
end

local bodyTag = {}
local flushPending

local lastBody, lastBodyPid = nil, -1

local function tagBody(body, pid)
    if not body or body == 0 then return end
    local k = kills[pid]
    if not k then return end

    local src = sourceContainer()
    if not src or src == 0 then
        lastErr = "no source text" ; return
    end

    local clone = safeCall(function()
        return Instantiate:CallStatic(src) end)
    if not clone or clone == 0 then
        lastErr = "instantiate failed" ; return
    end

    local ok = safeCall(function()
        local bt = GetTransform:Call(body)
        if not bt or bt == 0 then return false end
        local ct = GetTransform:Call(clone)
        if not ct or ct == 0 then return false end
        SetParent:Call(ct, bt, false)
        return true
    end)

    if not ok then
        lastErr = "parent failed" ; return
    end

    safeCall(function() SetActive:Call(clone, true) end)

    safeCall(function()
        local tmp = GetComp:Call(clone, "TextMeshPro")
        if not tmp or tmp == 0 then
            tmp = GetComp:Call(clone, "TMP_Text")
        end
        if tmp and tmp ~= 0 then
            SetText:Call(tmp, labelFor(k, os.clock()))
            bodyTag[pid] = tmp
            tagged = tagged + 1
        else
            lastErr = "no TMP in clone"
        end
    end)
end

    _G.HookBus.on("FixedUpdate", function(instance)
                local r = nil

                pcall(function()
                    tick = tick + 1
                    if tick < 30 then return end
                    tick = 0

                    local ss = SS_Field and safeCall(function()
                        return SS_Field:GetStatic() end) or nil
                    local inGame = (ss ~= nil and ss ~= 0)

                    if inGame and not wasIn then
                        wasIn = true
                        for k in pairs(kills) do kills[k] = nil end
                        for k in pairs(bodyTag) do bodyTag[k] = nil end
                        nKills, tagged = 0, 0
                        status = "game started"
                    elseif not inGame and wasIn then
                        wasIn = false
                        for k in pairs(bodyTag) do bodyTag[k] = nil end
                        status = "game ended"
                    end

                    if not inGame or not S.onBody then return end

                    flushPending()

                    if lastBody ~= nil and lastBodyPid >= 0 then
                        local b, p = lastBody, lastBodyPid
                        lastBody, lastBodyPid = nil, -1
                        if not bodyTag[p] and kills[p] then
                            tagBody(b, p)
                        end
                    end

                    if S.showTime and SetText then
                        local now = os.clock()
                        for pid, tmp in pairs(bodyTag) do
                            local k = kills[pid]
                            if k and tmp and tmp ~= 0 then
                                safeCall(function()
                                    SetText:Call(tmp, labelFor(k, now))
                                end)
                            end
                        end
                    end
                end)

    end)
end

local function getDevice(info)
    if not info or info == 0 then
        return "?"
    end

    local instance = getClientInstance()
    if not instance then
        return "?"
    end

    local clientData = nil

    if getClientFromInfoCall then
        local ok = pcall(function()
            clientData = getClientFromInfoCall:Call(instance, info)
        end)
        if not ok then
            clientData = nil
        end
    end

    if (not clientData or clientData == 0) and getClientCall then
        local id = getField(clientIdField, info)
        if id ~= nil then
            local ok = pcall(function()
                clientData = getClientCall:Call(instance, id)
            end)
            if not ok then
                clientData = nil
            end
        end
    end

    if not clientData or clientData == 0 then
        return "?"
    end

    local platformData = getField(platformDataField, clientData)
    if not platformData or platformData == 0 then
        return "?"
    end

    local platform = getField(platformField, platformData)
    return devices[tonumber(platform)] or "?"
end

local function getColorHex(info)
    if not NPI_GetOutfit or not info or info == 0 then
        return nil
    end

    local success, hex = pcall(function()
        local outfit = NPI_GetOutfit:Call(info)

        if not outfit or outfit == 0 then
            return nil
        end

        local colorId =
            BOMBA.read_int(outfit, OUTFIT_COLOR_OFF)

        if colorId and colorId >= 18 then
            return "#123F1B"
        end

        return COLOR_HEX[colorId]
    end)

    return success and hex or nil
end

local function getRoleInfo(info)
    if not info or info == 0 then
        return nil, false
    end

    local success, roleName, impostor = pcall(function()
        local rolePtr =
            BOMBA.read_ptr(info, NPI_ROLE_OFF)

        if not rolePtr or rolePtr == 0 then
            return nil, false
        end

        local name =
            getNiceName:Call(rolePtr)

        if not name or name == "" then
            return nil, false
        end

        return name, getIsImpostor:Call(rolePtr) == true
    end)

    if success then
        return roleName, impostor
    end

    return nil, false
end

function BUS.on(evt, fn)
    if type(fn) ~= "function" then return end
    BUS.subs[evt] = BUS.subs[evt] or {}
    local list = BUS.subs[evt]
    list[#list + 1] = fn
end

local function fire(evt, ...)
    local list = BUS.subs[evt]
    if not list then return end
    for i = 1, #list do
        pcall(list[i], ...)
    end
end

local function loadModule(name)
    local s = SRC[name]
    if not s then loaded[name] = { ok = false, msg = "no source" } ; return end

    local prevDraw = DrawImGui
    DrawImGui = nil
    local captured = nil

    ImGui = setmetatable({
        SetDrawCallback   = function(fn) captured = fn end,
        ClearDrawCallback = function() captured = nil end,
    }, { __index = realImGui })

    local chunk, err = load(s, "@" .. name)
    if not chunk then
        ImGui = realImGui ; DrawImGui = prevDraw
        loaded[name] = { ok = false, msg = "syntax", failed = true }
        log("[!] " .. name .. " syntax")
        return
    end

    local ok, ret = pcall(chunk)
    local md = DrawImGui or captured
    ImGui = realImGui
    DrawImGui = prevDraw

    if not ok then
        loaded[name] = { ok = false, msg = tostring(ret), failed = true }
        log("[!] " .. name .. " failed")
        return
    end

    if md then drawOf[name] = md end
    loaded[name] = { ok = true, ui = md ~= nil }
    log("[+] " .. name)
end
